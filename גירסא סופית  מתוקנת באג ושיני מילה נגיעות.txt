// ===== ANALYTICS ENGINE =====
const STORAGE_KEY = 'aviram_analytics';

function loadData() {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY)) || { visits: 0, phone: 0, wa: 0, events: [] };
  } catch(e) {
    return { visits: 0, phone: 0, wa: 0, events: [] };
  }
}

function saveData(data) {
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(data)); } catch(e) {}
}

function logEvent(type, label) {
  const data = loadData();
  if (type === 'visit') data.visits++;
  if (type === 'phone') data.phone++;
  if (type === 'wa') data.wa++;
  const now = new Date();
  const timeStr = now.toLocaleTimeString('he-IL', { hour: '2-digit', minute: '2-digit' });
  const dateStr = now.toLocaleDateString('he-IL', { day: '2-digit', month: '2-digit' });
  data.events.unshift({ label, time: timeStr + ' ' + dateStr });
  if (data.events.length > 20) data.events = data.events.slice(0, 20);
  saveData(data);
  refreshDash();
}

function refreshDash() {
  const data = loadData();
  document.getElementById('stat-visits').textContent = data.visits;
  document.getElementById('stat-phone').textContent = data.phone;
  document.getElementById('stat-wa').textContent = data.wa;
  const list = document.getElementById('events-list');
  if (data.events.length === 0) {
    list.innerHTML = '<p style="color:#aaa;font-size:13px;">עדיין אין פעולות מוקלטות</p>';
  } else {
    list.innerHTML = data.events.map(e =>
      `<div class="event-item"><span>${e.label}</span><span class="event-time">${e.time}</span></div>`
    ).join('');
  }
}
